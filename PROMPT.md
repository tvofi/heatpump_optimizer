# Counts planned hot-water runs that begin on a tank at or above setpoint - h (the pump's start threshold).
import os, sys
sys.path.insert(0, "tests"); sys.path.insert(0, "custom_components")
import numpy as np
from datetime import datetime
from profiles import prices, weather, house, solve_inputs
from heatpump_optimizer.thermal_model import ThermalModel, ThermalParameters, ThermalState
from heatpump_optimizer.optimizer import HeatPumpOptimizer, OptimizationConfig
START = datetime(2026,1,15,0,0)
def runs(price_p, weather_p, tank0):
    cfg = house(two_zone=False, dhw=True)
    p = ThermalParameters.from_config(cfg); p.dhw_enabled = True
    oc = OptimizationConfig(horizon_hours=24, time_step_minutes=15, target_temp=cfg["target_temperature"], min_temp=cfg["min_temperature"], max_temp=cfg["max_temperature"])
    opt = HeatPumpOptimizer(ThermalModel(p), oc)
    ot, wind, rain, sol = weather(weather_p, START)
    st = ThermalState(room_temperature=21.0, slab_temperature=22.0, outdoor_temperature=float(ot[0]), dhw_temperature=tank0, dhw_hours_since_legionella=20.0, upper_floor_temperature=21.0, lower_floor_temperature=21.0, buffer_tank_temperature=40.0)
    r = opt.optimize(inputs=solve_inputs(initial_state=st, prices=prices(price_p, START), outdoor_temps=ot, wind_speeds=wind, precipitation=rain, solar_radiation=sol, start_time=START))
    pw = np.asarray(r.dhw_power_schedule) > 1e-6; T = np.asarray(r.dhw_temp_trajectory)
    out = []
    for i in range(len(pw)):
        if pw[i] and (i == 0 or not pw[i-1]):
            j = i
            while j < len(pw) and pw[j]: j += 1
            out.append((i, round(float(T[i]),1), j-i, round(float(np.asarray(r.dhw_power_schedule)[i:j].sum()*0.25),2)))
    return p.dhw_setpoint, out

tot=0; warm={2:0,4:0,6:0}; kwh={2:0.0,4:0.0,6:0.0}; allk=0.0
for pr_ in ("winter_typical","flat","negative_night","evening_spike"):
    for we_ in ("winter_cold","winter_mild","summer_warm"):
        for t0 in (45.0, 50.0, 53.0):
            try: sp, rs = runs(pr_, we_, t0)
            except Exception as e:
                print("skip", pr_, we_, type(e).__name__, str(e)[:80]); break
            for (i,T,n,k) in rs:
                if i == 0: continue
                tot += 1; allk += k
                for h in warm:
                    if T >= sp - h: warm[h] += 1; kwh[h] += k
            print(pr_, we_, t0, "sp", sp, [(i,T,n) for (i,T,n,k) in rs])
print("RUNS not at step 0:", tot, "kWh", round(allk,2))
for h in warm: print(f"start at/above setpoint-{h}K: {warm[h]} runs, {kwh[h]:.2f} kWh")
