OLD = "if take <= 1e-6:"
NEW = "if take < 1e-6:"
MUT = mutant_method("_plan_dhw_cheapest_first", OLD, NEW)
from dataclasses import replace  # noqa: E402
from custom_components.heatpump_optimizer.dhw_planner import _DhwPlanContext  # noqa: E402
from custom_components.heatpump_optimizer.thermal_model import ThermalModel, ThermalParameters  # noqa: E402
from custom_components.heatpump_optimizer.optimizer import OptimizationConfig  # noqa: E402

PARAMS = ThermalParameters(dhw_enabled=True, dhw_tank_volume=200.0, dhw_cooling_rate=0.3)
PLANNER = DhwPlanner(ThermalModel(PARAMS), OptimizationConfig(horizon_hours=12, time_step_minutes=60), None, None)


def ctx(prices, lead, n, c_dhw=0.232):
    prices = np.asarray(prices, dtype=float)
    return _DhwPlanContext(prices=prices, import_prices=prices, outdoor_temps=np.full(n, -5.0), humidity=None,
                           n_steps=n, dt=1.0, store=replace(PARAMS.dhw_store, capacity_kwh_per_c=c_dhw),
                           p_dhw_max=3.0, min_run_power=0.1, max_lead_steps=lead, tank_exo_kw=None)


def greedy(prices, lead, req, max_temp, t0=45.0, draws=None, forced_off=None, c_dhw=0.232, fn=None):
    """The plan DhwPlanner._plan_dhw_cheapest_first returns (kW per step)."""
    n = len(prices)
    return (fn or DhwPlanner._plan_dhw_cheapest_first)(
        PLANNER, ctx(prices, lead, n, c_dhw), t0, np.asarray(req, dtype=float).copy(),
        np.zeros(n) if draws is None else np.asarray(draws, dtype=float), np.asarray(max_temp, dtype=float).copy(),
        forced_off=forced_off)


N = 12
CHEAP_EARLY = np.array([0.1] + [5.0] * 11)        # the cheapest hour is 10 h before the need
REQ = np.zeros(N)
REQ[10] = 55.0                                    # hot water needed at 10:00
MAX_T = np.full(N, 65.0)

# Construct the tie take == 1e-6 exactly: a 6-step horizon from a 60 C tank drawn down below a 58 C requirement at
# step 4; only step 0 may run (steps 1-5 pinned off), and the ceiling at step 1 leaves a headroom d (in K) whose
# d * c_dhw rounds to exactly 1e-6 kWh. The tank trajectory comes from the model's own store, so c_dhw (the
# planner's divisor) can be tuned to the tie without moving the temperatures.
n = 6
draws = np.array([0, 0, 0.6, 0.6, 0, 0])
req = np.zeros(n)
req[4] = 58.0
fo = np.array([False] + [True] * 5)
temps = PLANNER._dhw_plan_temps(np.zeros(n), 60.0, np.full(n, 5.0), draws, 1.0, None, None)
t2 = float(temps[2])
x = t2 + 1.0 + 1e-6 / 0.232
d = float((x - 1.0) - t2)
c_tie = None
for direction in (np.inf, -np.inf):
    y = 1e-6 / d
    for _ in range(200):
        if d * y == 1e-6:
            c_tie = y
            break
        y = np.nextafter(y, direction)
    if c_tie is not None:
        break
mt = np.full(n, 90.0)
mt[1] = x
prices = np.ones(n)


def g(max_temp, c, fn=None):
    return (fn or DhwPlanner._plan_dhw_cheapest_first)(
        PLANNER, replace(ctx(prices, 24, n, c), outdoor_temps=np.full(n, 5.0)), 60.0, req.copy(), draws,
        max_temp.copy(), forced_off=fo)


b, m = g(mt, c_tie), g(mt, c_tie, fn=MUT)
print(f"# tie: d={d!r} K c_dhw={c_tie!r} d*c_dhw={d * c_tie!r}")
print(f"RESULT step0_kw_baseline={float(b[0]):.4f} kW")
print(f"RESULT step0_kw_mutant={float(m[0]):.4f} kW")
offs = {"headroom_+1e-9K": (x + 1e-9, c_tie), "headroom_-1e-9K": (x - 1e-9, c_tie), "c_dhw_0.232": (x, 0.232),
        "loose_ceiling": (90.0, 0.232)}
moved = []
for name, (xv, c) in offs.items():
    mm = mt.copy()
    mm[1] = xv
    if not np.array_equal(g(mm, c), g(mm, c, fn=MUT)):
        moved.append(name)
print(f"RESULT off_tie_inputs_moved={len(moved)} of {len(offs)} ({','.join(moved) or '-'})")
footer()
