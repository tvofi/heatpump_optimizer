def results(path):
    out = {}
    for line in open(path):
        if line.startswith("RESULT "):
            for m in re.finditer(r"(\w+)=(\S+)", line[7:]):
                out.setdefault(m.group(1), m.group(2))
            if "inputs_tried=" in line:
                out["inputs_tried"] = line.split("inputs_tried=", 1)[1].strip()
            if "reach=" in line:
                out["reach"] = line.split("reach=", 1)[1].strip()
    return out
