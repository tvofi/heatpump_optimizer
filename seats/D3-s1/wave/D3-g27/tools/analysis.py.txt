def probe_results(key):
    f = W / "probes" / f"probe_{key}.py"
    return [l[len("# EXPECTED RESULT "):].strip() for l in f.read_text().splitlines() if l.startswith("# EXPECTED RESULT ")]


def get(res, name):
    for r in res:
        if r.startswith(name + "="):
            return r.split("=", 1)[1]
    return None


def finalize(key):
    keys = {l.split("\t")[0]: l.split("\t") for l in (W / "keys.tsv").read_text().splitlines()[1:] if l.strip()}
    row = keys[key]
    g = json.loads((W / "gates" / f"{key}.json").read_text())
    mut = json.loads((S1 / "state/results/complete" / f"{key}.json").read_text())["mutant"]
    a = A[key]
    res = probe_results(key)
    cmd = f"{PY} {WREL}/probes/probe_{key}.py"
    gate = {k: g.get(k) for k in ("verdict", "killed_by", "gate_rc", "wall_s", "load1_start", "load1_end")}
    gate["record"] = f"{WREL}/gates/{key}.json"
    gate["command"] = f"{PY} dev/audit/rounds/round11/D3/s1/fullgate.py run --src {row[1]} --key {key}"
    gate["provisional"] = True
    out = {"key": key, "site": f"{mut['file'].split('/')[-1]}:{mut['line']} {mut['kind']}",
           "patch": f"{mut['old'].strip()} -> {mut['new'].strip()}", "function": row[4], "cluster": row[5],
           "slice": row[6], "m4_source": row[10], "gate": gate}
    v = g.get("verdict")
    if v == "KILLED":
        out["class"] = "killed"
        out["killed_by"] = g.get("killed_by")
        out["fail_lines"] = g.get("fail_lines")
        out["non_finding"] = {"claim": f"{out['site']} {key} is pinned by the suite: the full gate KILLED it",
                              "command": gate["command"],
                              "value": f"killed_by={','.join(g.get('killed_by') or [])}; gate_rc={g.get('gate_rc')}; wall {g.get('wall_s')} s (provisional)"}
        out["probe"] = {"path": f"{WREL}/probes/probe_{key}.py", "command": cmd, "results": res,
                        "note": "probe taken before the verdict; not needed for a killed key"}
        return out
    if v == "TIMEOUT" or v is None:
        out["class"] = "timeout"
        log = W / "gates" / f"{key}.log"
        last = None
        if log.exists():
            ran = re.findall(r"(tests/[\w./-]+\.(?:py|mjs|sh))", log.read_text(errors="replace"))
            last = ran[-1] if ran else None
        out["timeout"] = {"cap_s": g.get("cap_s", 14400), "last_script": last}
        return out
    moved = a.get("title") is not None
    out["class"] = "candidate" if moved else "equivalent-candidate"
    out["probe"] = {"path": f"{WREL}/probes/probe_{key}.py", "command": cmd,
                    "baseline_value": get(res, "h_baseline") or get(res, "bounds_baseline") or get(res, "kept_baseline") or get(res, "output_baseline"),
                    "mutant_value": get(res, "h_mutant") or get(res, "bounds_mutant") or get(res, "kept_mutant") or get(res, "output_mutant"),
                    "example_input": get(res, "example_input"),
                    "results": res,
                    "inputs_tried": f"production domain: {a['domain']}; " + "; ".join(r for r in res if "_tried=" in r),
                    "null": [r for r in res if r.startswith("null_")]}
    out["step4"] = a["reason"]
    out["should_fail"] = a["should_fail"]
    out["machine"] = "box D3-g27: 4 vCPU Intel Xeon @ 2.10GHz, 16 GB, Linux, venv-ci CPython 3.14"
    if moved:
        out["finding"] = {
            "title": a["title"], "claim": a["claim"] + f" Full gate: {v}, gate_rc={g.get('gate_rc')}.",
            "step": "D3.M3", "scope": "D3-s1", "class_guess": a["class_guess"], "severity_claimed": a["severity"],
            "stop_rule_class": "bug", "metric_definition": a["metric"], "phenomenon_property": a["phenomenon"],
            "seam_rule": "awk -F'\\t' '$6==\"gate\"' SURVIVORS.tsv (the wave list; this key's row) with the full gate's verdict in gates/<key>.json",
            "instrumented_symbol": f"custom_components.heatpump_optimizer.optimizer:{row[4]}",
            "files": [mut["file"]],
            "harness_paths": ["dev/audit/rounds/round11/D3/s1/fullgate.py", f"{WREL}/probes/probe_{key}.py"],
            "evidence_command": cmd,
            "evidence": {"command": cmd, "harness_path": f"{WREL}/probes/probe_{key}.py", "value": a["value"], "unit": a["unit"],
                         "baseline_sha": "92d38ecb640bd462f2e19c0814d0f0b7df1ee692", "machine": out["machine"],
                         "tolerance": "exact (counts); solver floats to 1e-3 relative", "cpu_or_wall": "n/a"},
            "perturbation": {"change": f"the one-line mutant `{mut['old'].strip()}` -> `{mut['new'].strip()}` at {mut['file']}:{mut['line']}, in memory by the probe; on disk by fullgate.py run --key {key}",
                             "expected_direction": "up", "observed_value": a["value"]},
            "null_control": {"command": cmd, "value": "; ".join(r for r in res if r.startswith("null_")),
                             "note": "the function re-compiled with the line unchanged: no grid input, production call or plan moves; the box's full-gate null is gates/null.json"},
            "should_fail": a["should_fail"],
            "proposed_fix_scope": "tests only: the check named in should_fail, at the input the probe's example_input names",
        }
    else:
        out["non_finding"] = {"claim": f"{out['site']} {key} survives the full gate but is equivalent at every production-reachable input tried: {a['reason']}",
                              "command": cmd, "value": "; ".join(r for r in res if "moved" in r or "changed" in r)}
    return out


if __name__ == "__main__":
    for key in sys.argv[1:]:
        r = finalize(key)
        (W / "results" / f"{key}.json").write_text(json.dumps(r, indent=1) + "\n")
        print(key, r["class"])
