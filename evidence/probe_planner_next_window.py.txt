# PROBE-START reviewer probe: the planner's published next-window hours on a weekly spec
def _probe_next(spec, start, h0):
    _cfg = _grad_house(two_zone=False, dhw=True)
    _cfg["dhw_windows"] = spec
    _p = ThermalParameters.from_config(_cfg)
    _m = ThermalModel(_p)
    _o = _PvOpt(_m, _PvOptCfg(horizon_hours=24, time_step_minutes=15, target_temp=21.0, min_temp=17.0, max_temp=23.0))
    _n = 96
    import datetime as _dtm
    t0 = _WDT.combine(start.date(), _WDT.min.time()) + _dtm.timedelta(hours=h0)
    _ts = [t0 + _dtm.timedelta(hours=i * 0.25) for i in range(_n)]
    _hours = np.array([t.hour + t.minute / 60.0 for t in _ts])
    _plan, _ = _dhw_planner(_o)._build_dhw_requirements(
        _dhw_horizon(prices=np.full(_n, 1.0), outdoor=np.full(_n, -5.0), n=_n, dt=0.25, step_hours=_hours,
            step_weekdays=np.array([(t0 + _dtm.timedelta(hours=(i - 1) * 0.25)).weekday() for i in range(_n + 1)])),
        p_max=4.0)
    return _plan.next_window_in_hours, _o
