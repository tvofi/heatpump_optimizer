import json,re,sys,pathlib
root=pathlib.Path(sys.argv[1])/"custom_components/heatpump_optimizer"
def leaves(d,p=""):
    if isinstance(d,dict):
        for k,v in d.items(): yield from leaves(v,f"{p}.{k}" if p else k)
    else: yield p,d
ph=lambda s: sorted(set(re.findall(r"\{[^{}]+\}",s))) if isinstance(s,str) else []
files={"strings":root/"strings.json",**{p.stem:p for p in (root/"translations").glob("*.json")}}
L={n:dict(leaves(json.loads(p.read_text()))) for n,p in files.items()}
ref=L["strings"]; bad=0
for n,d in L.items():
    miss=sorted(set(ref)-set(d)); extra=sorted(set(d)-set(ref))
    phx=[k for k in ref if k in d and ph(ref[k])!=ph(d[k])]
    same=[k for k in ref if n not in("strings","en") and k in d and d[k]==L["en"].get(k) and isinstance(d[k],str) and len(d[k])>12]
    print(f"RESULT {n}: leaves={len(d)} missing={len(miss)} extra={len(extra)} placeholder_mismatch={len(phx)}", miss[:5], extra[:5], phx[:5])
    bad+=len(miss)+len(extra)+len(phx)
print("RESULT parity_total_defects", bad); sys.exit(1 if bad else 0)
