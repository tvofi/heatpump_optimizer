# usage: mut.py <label> <file> <old> <new> -- <cmd...>  ; run in wt-head, restore after
import subprocess, sys, pathlib, os
label, f, old, new = sys.argv[1:5]; cmd = sys.argv[sys.argv.index("--")+1:]
p = pathlib.Path(f); src = p.read_text()
n = src.count(old)
if n != 1: print(f"MUTANT {label}: old text found {n} times, not applied"); sys.exit(9)
p.write_text(src.replace(old, new))
try:
    r = subprocess.run(cmd, capture_output=True, text=True, timeout=1500, env={**os.environ, "PYTHONPATH": "tests/hastub"})
    out = r.stdout + r.stderr
    fails = [l for l in out.splitlines() if "FAIL" in l][:6]
    print(f"MUTANT {label}: rc={r.returncode} {'KILLED' if r.returncode else 'SURVIVED'}")
    for l in fails: print("   ", l[:220])
finally:
    p.write_text(src)
