import json, subprocess, sys
sys.path[:0] = ["tools/audit", "tools/audit/seat"]
import carry_audit, roster_lib
ref = sys.argv[1]
roster = json.loads(subprocess.run(["git","show",f"{ref}:dev/programme/rosters/wave-r10-groups.json"],capture_output=True,text=True,check=True).stdout)
d = carry_audit.Dests(roster, set())
G = roster_lib.groups_of(roster)
bypass = {}
for n, owner in d.owned.items():
    g = G.get(owner)
    if g is not None and not roster_lib.carry_receivable(g):
        bypass[n] = (owner, (g.get("resume") or {}).get("stage"))
for n,(o,s) in sorted(bypass.items()):
    ok, note = d.check(f"#{n}")
    print(f"#{n} owner={o} stage={s} accepted_without_state={ok}")
print(f"RESULT issue_owned_by_unreceivable={len(bypass)}")
