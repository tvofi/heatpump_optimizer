import subprocess, sys, pathlib
P = pathlib.Path("tools/audit/check_scopes.py"); orig = P.read_text()
M = {
 "NULL": ("", ""),
 "M1 isinstance->False": ("if not isinstance(gating, bool):", "if False:"),
 "M2 isinstance->is None": ("if not isinstance(gating, bool):", "if gating is None:"),
 "M3 drop merged guard": ("elif not gating and not (c and merged(c)):", "elif not gating:"),
 "M4 elif False": ("elif not gating and not (c and merged(c)):", "elif False:"),
 "M5 no append": ("                parallel.append(cid)\n", "                pass\n"),
 "M6 no continue": ("                parallel.append(cid)\n            continue\n", "                parallel.append(cid)\n"),
 "M7 old norm": ('(norm(ev[k]) if full(ev.get(k)) else "" for k in', '(norm(ev.get(k, "")) for k in'),
 "M8 gating is True only (not gating->gating is False)": ("elif not gating and", "elif gating is False and"),
 "M9 non-gating unmerged still gap": ("            continue\n        if not (c and merged(c)):", "        if not (c and merged(c)):"),
 "M10 open-cards rc 2->0": ("            sys.exit(2)\n        out = []", "            sys.exit(0)\n        out = []"),
}
for name, (a, b) in M.items():
    if a and orig.count(a) != 1: print(f"{name}: SPEC-MISS count={orig.count(a)}"); continue
    P.write_text(orig.replace(a, b) if a else orig)
    r = subprocess.run([sys.executable, sys.argv[1]], capture_output=True, text=True)
    print(f"{name}: {'KILLED' if r.returncode else 'SURVIVED'}  {r.stdout.strip().splitlines()[0] if r.stdout.strip() else r.stderr[-200:]}")
P.write_text(orig)
print("restored", subprocess.run(["git","diff","--stat"],capture_output=True,text=True).stdout or "clean")
