import subprocess, sys, pathlib
P = pathlib.Path("tools/audit/check_scopes.py"); orig = P.read_text()
M = {
 "M10 open-cards rc 2->0": ("            sys.exit(2)\n        out = []", "            sys.exit(0)\n        out = []"),
 "M11 open-cards drops list": ("evaluation_gaps(a.round, tree, show, merged, anc, changed, out)", "evaluation_gaps(a.round, tree, show, merged, anc, changed)"),
}
for name, (a, b) in M.items():
    assert orig.count(a) == 1, name
    P.write_text(orig.replace(a, b))
    r = subprocess.run([sys.executable, "tools/audit/check_scopes.py", "--repo", ".", "--ref", "HEAD", "--self-test"], capture_output=True, text=True)
    line = [l for l in r.stdout.splitlines() if "evaluation arms" in l]
    print(f"{name}: rc={r.returncode} {'KILLED' if r.returncode else 'SURVIVED'} {line}")
P.write_text(orig)
