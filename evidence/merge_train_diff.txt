+def stop_on_signal() -> None:
+    """TERM and HUP raise SystemExit(128 + signal), so a stop unwinds through
+    Train._run's handler and the train's own `finally` blocks."""
+    def unwind(signum, _frame):
+        raise SystemExit(128 + signum)
