+def _padded(values: np.ndarray | None, n: int) -> np.ndarray:
+    """``values`` as ``n`` floats, NaN past its end, so no comparison holds there."""
+    out = np.full(n, np.nan)
+    if values is not None:
+        m = min(n, len(values))
+        out[:m] = np.asarray(values[:m], dtype=float)
+    return out
