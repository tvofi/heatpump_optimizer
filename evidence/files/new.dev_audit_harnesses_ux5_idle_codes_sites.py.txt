+"""R9-UX-5: the six idle_codes mutation sites #2010's `mutation` left unpinned, measured on real trees.
+Each variant is a `git archive` of a commit with one line replaced, imported
+fresh, and driven through the REAL `idle_codes` and its two callers.
+  A. simplify: PREV (the head before the redundant solar terms were removed)
+     against HEAD over a random grid of 1-D channels (lengths that disagree,
+     NaN, +-inf, the 0.05 threshold and the 1e-6 surplus cut, absent arrays),
+     every step, `idle_codes` and both classifiers. Control: HEAD with the
+     solar bound `nxt_sun < nxt_run` moved to `<=` must differ.
+  B. kills: each HEAD mutant the `tests/features.py` UX-5 checks must kill, on
+     those checks' own inputs; the restored HEAD on the same inputs.
+  C. `n <= 0` -> `n < 0` (the triaged CMP_BOUND): HEAD against the mutant at
+     n in {-2..3} over the same grid, outputs and exception types compared.
+     Control: the mutant plus a body that answers ["n0"] for an empty plan --
+     it differs only where the body runs at n == 0, so a probe that never
+     reaches the boundary prints 0 there.
+    PYTHONPATH=tests/hastub python3 dev/audit/harnesses/ux5_idle_codes_sites.py PREV HEAD [rounds]
+PREV is 217f7cefe95848c6d22f3172e61ab6419fb0b316 and HEAD
+57cac0889c2af50fdf4205839710eead4d0ee3e6 (or any later commit with the same
+`idle_codes`); rounds defaults to 4000.
+"""
+from __future__ import annotations
+import random
+import subprocess
+import sys
+import tempfile
+from pathlib import Path
+def repo_root(start):
+    """The directory holding custom_components/heatpump_optimizer/manifest.json."""
+    from pathlib import Path
+    here = Path(start).resolve()
+    if here.is_file():
+        here = here.parent
+    marker = Path("custom_components") / "heatpump_optimizer" / "manifest.json"
+    for cand in (here, *here.parents):
+        if (cand / marker).is_file():
+            return cand
+    raise RuntimeError(f"no repository root above {start}")
+REPO = repo_root(__file__)
+PREV, HEAD = sys.argv[1], sys.argv[2]
+ROUNDS = int(sys.argv[3]) if len(sys.argv) > 3 else 4000
+OPT = "custom_components/heatpump_optimizer/optimizer.py"
+_TMP = tempfile.TemporaryDirectory(prefix="ux5sites-")
+TMP = Path(_TMP.name)
+def tree(ref, name, edits=()):
+    d = TMP / name
+    d.mkdir()
+    a = subprocess.run(["git", "-C", str(REPO), "archive", ref, "custom_components", "tests/hastub"],
+                       check=True, capture_output=True).stdout
+    subprocess.run(["tar", "-x", "-C", str(d)], input=a, check=True)
+    p = d / OPT
+    s = p.read_text()
+    for old, new in edits:
+        assert s.count(old) == 1, (name, old)
+        s = s.replace(old, new)
+    p.write_text(s)
+    return d
+def load(root):
+    for k in [k for k in sys.modules if k.startswith("heatpump_optimizer")]:
+        del sys.modules[k]
+    sys.path[:0] = [str(root / "tests" / "hastub"), str(root / "custom_components")]
+    try:
+        import heatpump_optimizer.optimizer as m
+    finally:
+        del sys.path[:2]
+    return m
+import numpy as np
+SOLAR = "        codes[~(sp > 1e-6) & (nxt_sun < nxt_run)] = REASON_IDLE_SOLAR\n"
+GUARD = "    if n <= 0:\n"
+SUNNY = "        sunny = np.flatnonzero(np.asarray(surplus, dtype=float) > 1e-6)\n"
+RET = "    return list(codes.tolist())\n"
+def grid(seed, rounds, ns=None):
+    rng = random.Random(seed)
+    def arr(n, temps=False):
+        if rng.random() < 0.15:
+            return None
+        out = []
+        for _ in range(max(0, n + rng.choice([-2, -1, 0, 0, 0, 1, 3]))):
+            r = rng.random()
+            out.append(float("nan") if r < 0.03 else float("inf") if r < 0.04
+                       else float("-inf") if r < 0.05 else 0.0 if r < 0.35
+                       else 0.05 if r < 0.45 else 1e-6 if r < 0.5
+                       else round(rng.uniform(0, 30 if temps else 3), rng.choice([0, 1, 2])))
+        return np.array(out, dtype=float)
+    for _ in range(rounds):
+        n = rng.choice(ns) if ns else rng.randint(0, 14)
+        power = arr(max(n, 0))
+        power = np.zeros(max(n, 0)) if power is None else power
+        yield (n, power, arr(max(n, 0)), arr(max(n, 0) + 1, True), arr(max(n, 0), True),
+               arr(max(n, 0)), arr(max(n, 0)), arr(max(n, 0)), rng.choice([0.05, 0.05, 0.0, 1.0]))
+def run(m, case):
+    n, power, prices, level, floor, surplus, other, caps, thr = case
+    out = []
+    for f in (lambda: m.idle_codes(n, power, prices, level, floor, surplus, other, caps, thr),
+              lambda: m.classify_dhw_steps(power, np.zeros(max(n, 0), bool), np.zeros(max(n, 0)),
+                                           None, n, thr, m.IdleContext(prices=prices, level=level,
+                                           floor=floor, surplus=surplus, other=other, caps=caps)),
+              lambda: (m.classify_space_steps(power, prices, level, floor, np.ones(n), surplus, n, thr,
+                                              other=other, caps=caps)
+                       if prices is not None and len(prices) >= n and level is not None and len(level)
+                       and floor is not None and len(floor) >= n and n >= 0 else None)):
+        try:
+            v = f()
+            out.append(repr(v) + repr([type(x) for x in v] if isinstance(v, list) else ""))
+        except Exception as e:
+            out.append("EXC " + type(e).__name__)
+    return out
+def diff(a, b, cases):
+    return sum(run(a, c) != run(b, c) for c in cases)
+prev, head = load(tree(PREV, "prev")), load(tree(HEAD, "head"))

+checks = {

+        ((3, np.array([0.0, 0.0, 2.0]), None, None, None, np.array([0.0, 1e-6, 0.0]), None, None, T), "idle"),
+    "surplus that arrives with the next run is not a wait":
+        ((2, np.array([0.0, 1.0]), None, None, None, np.array([0.0, 2.0]), None, None, T), "idle"),
+    "a negative step count has no idle codes":
+        ((-1, np.zeros(1), None, None, None, None, None, None, T), []),

+mutants = {
