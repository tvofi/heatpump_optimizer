"""Reviewer-built harness (not the finder's; the body names none): drives the three
orderers at a given tree with fixtures and the live roster; prints RESULT lines."""
import sys, json, copy, datetime as dt, tempfile, random, importlib
from pathlib import Path
T = Path(sys.argv[1]); tag = sys.argv[2]
sys.path[:0] = [str(T/"tools/plan"), str(T/"tools/audit/seat")]
import replan, open_driver as od, merge_train as mt, roster_lib
head = hasattr(roster_lib, "is_outside")
M = replan.MEASURED_FIXTURE; start = dt.datetime(2026, 1, 5)
caps = dict(replan.DEFAULT_CAPS, slots=1, seats=8)
def grp(n, size="S", kind="defect", after=(), **kw):
    return {"group": n, "size": size, "kind": kind, "after": list(after), "resume": {"stage": "not-started"}, **kw}
def seq(ros):
    p = replan.build_plan(ros, M, caps, start, "v1.0.0", round_n=2)
    return sorted(p["groups"], key=lambda n: (p["groups"][n]["start"], n))
# 1 replan: outside / hotfix / cp
ros = {"groups": [grp("CPA", "L", "instrument"), grp("CPB", "S", "instrument", after=["CPA"]),
                  grp("R10-HOTFIX-H", "S", "bug"), grp("OUT", "S", "instrument", source="outside"), grp("PROD", "M")]}
print(f"RESULT {tag} replan seq with outside: {seq(ros)}")
nul = {"groups": [{k: v for k, v in g.items() if k != "source"} for g in ros["groups"]]}
print(f"RESULT {tag} replan seq null (no source): {seq(nul)}")
nul2 = {"groups": [g for g in nul["groups"] if g["group"] != "R10-HOTFIX-H"]}
print(f"RESULT {tag} replan seq null (no source, no hotfix): {seq(nul2)}")
# random null control: no source field, no hotfix -> base identical
random.seed(7); outs = []
for i in range(60):
    gs = []
    for j in range(7):
        k = random.choice(["defect", "instrument", "instrument"])
        aft = random.sample([g["group"] for g in gs], k=min(len(gs), random.randint(0, 2)))
        kw = {"savings": random.choice([None, 3, 15])} if k == "instrument" else {}
        kw = {a: b for a, b in kw.items() if b is not None}
        gs.append(grp(f"G{j}", random.choice("SML"), k, aft, **kw))
    outs.append(seq({"groups": gs}))
print(f"RESULT {tag} replan random-null digest: {__import__("hashlib").sha256(json.dumps(outs).encode()).hexdigest()[:16]}")
# live roster
live = json.load(open(Path.home()/"live-roster.json"))
try:
    ls = replan.build_plan(copy.deepcopy(live), M, dict(replan.DEFAULT_CAPS), start, "v1.0.0", round_n=10)
    tiers = {n: ls["groups"][n].get("tier") for n in ls["groups"]}
    outside_live = [g["group"] for g in live["groups"] if g.get("source") == "outside"]
    print(f"RESULT {tag} live roster outside groups: {outside_live} tiers {[tiers.get(g) for g in outside_live]}")
    print(f"RESULT {tag} live roster tier histogram: {sorted(__import__('collections').Counter(tiers.values()).items(), key=str)}")
except Exception as e:
    print(f"RESULT {tag} live roster plan: ERROR {type(e).__name__}: {e}")
# 2 open_driver
items = [("r10-inst", "R10-INST", 1), ("r10-cp", "R10-CP", 2), ("triage-x", "R10-TRIAGE-X", 3),
         ("r10-hotfix-x", "R10-HOTFIX-X", 4), ("r10-out", "R10-OUT", 5)]
for pr in ([], ["r10-inst", "r10-out", "r10-cp", "r10-hotfix-x"], ["r10-out"] * 0 + ["r10-inst", "triage-x"]):
    if head:
        print(f"RESULT {tag} od order pr={pr}: {[t for t,_,_ in od.order(items, pr, {'r10-cp'}, {'r10-out'})]}")
    print(f"RESULT {tag} od order null pr={pr}: {[t for t,_,_ in od.order(items, pr, {'r10-cp'})]}")
noh = [i for i in items if not i[0].startswith("r10-hotfix")]
random.seed(3); digs = []
for i in range(300):
    it = random.sample(noh, k=random.randint(2, len(noh)))
    pr = random.sample([t for t,_,_ in noh] + ["zz"], k=random.randint(0, 4))
    digs.append([t for t,_,_ in od.order(it, pr, {"r10-cp"})])
print(f"RESULT {tag} od random-null (no outside, no hotfix) digest: {__import__("hashlib").sha256(json.dumps(digs).encode()).hexdigest()[:16]}")
# starvation: many outside + crit + listed priority, hotfix must be first
flood = [(f"r10-out{i}", f"R10-OUT{i}", i) for i in range(10)] + [("r10-hotfix-z", "R10-HOTFIX-Z", 99)]
if head:
    o = od.order(flood, [f"r10-out{i}" for i in range(10)], set(), {f"r10-out{i}" for i in range(10)})
    print(f"RESULT {tag} od flood: hotfix position {[t for t,_,_ in o].index('r10-hotfix-z')} of {len(o)}")
# 3 merge_train drain_order
heads = {1: "plain", 2: "cp", 3: "out", 4: "plain", 5: "hot"}
def run(argv, cwd=None, stdin=None):
    return 0, heads[int(argv[3])] + "\n"
ents = [{"pr": n, **({"priority": True} if n == 5 else {})} for n in (1, 2, 3, 4, 5)]
with tempfile.TemporaryDirectory() as d:
    kw = dict(run=run, log=lambda m: None, critical=lambda it: it["branch"] == "cp")
    if head: kw["outside"] = lambda it: it["branch"] == "out"
    t = mt.Train("o/r", Path(d), None, "", (), 1, **kw)
    print(f"RESULT {tag} drain with outside: {[e['pr'] for e in t.drain_order(copy.deepcopy(ents))]}")
    if head: kw["outside"] = lambda it: False
    t = mt.Train("o/r", Path(d), None, "", (), 1, **kw)
    print(f"RESULT {tag} drain null: {[e['pr'] for e in t.drain_order(copy.deepcopy(ents))]}")
    # live roster read path: no injection -> _critical/_outside at live roster (fixture via cp_read)
    kw2 = dict(run=run, log=lambda m: None)
    t = mt.Train("o/r", Path(d), None, "", (), 1, **kw2)
    t.cp_read = ({"groups": [{"group": "R10-O", "source": "outside", "resume": {"branch": "out", "pr": 3}},
                             {"group": "R10-C", "resume": {"branch": "cp", "pr": 2}}]}, {"r10-c"})
    print(f"RESULT {tag} drain via roster read: {[e['pr'] for e in t.drain_order(copy.deepcopy(ents))]}")
