for name, f, a, b in M:
    if only and name not in only: continue
    p = W/f; src = p.read_text()
    if src.count(a) != 1: print(f"RESULT mutant {name}: NOT-APPLIED (count {src.count(a)})", flush=True); continue
    p.write_text(src.replace(a, b))
    try:
        res = []
        for t in TESTS[f]:
            r = subprocess.run([sys.executable, t, "--self-test"], cwd=W, capture_output=True, text=True, timeout=600)
            fails = [l for l in (r.stdout+r.stderr).splitlines() if l.startswith(("FAIL", "  FAIL")) or "FAIL" in l[:8]]

        killed = any(rc != 0 for _, rc, _ in res)
        print(f"RESULT mutant {name}: {'KILLED' if killed else 'SURVIVED'} {[(t, rc) for t, rc, _ in res]}", flush=True)
        for t, rc, fl in res:
            for l in fl: print(f"    {t}: {l[:200]}", flush=True)
    finally:
        p.write_text(src)
