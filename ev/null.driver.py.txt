import sys, subprocess, json
sys.path.insert(0, "tools/audit")
import check_scopes as cs
REF = sys.argv[1]; mode = sys.argv[2]
run = lambda *a: subprocess.run(["git", *a], capture_output=True, text=True)
path = cs.records(11)[1]
def show(f, at=None):
    t = subprocess.check_output(["git", "show", f"{at or REF}:{f}"], text=True)
    if f == path and at is None and mode != "written":
        ev = json.loads(t)
        for c in ev.get("cards", {}).values():
            if c.get("decision") == "accepted": c["gates_finders"] = (mode == "true")
        t = json.dumps(ev)
    return t
merged = lambda sha: run("merge-base", "--is-ancestor", str(sha), REF).returncode == 0
anc = lambda x, y: run("merge-base", "--is-ancestor", str(x), y).returncode == 0
changed = lambda c: set(run("diff", "--name-only", f"{c}^1", str(c)).stdout.split())
tree = set(subprocess.check_output(["git", "ls-tree", "-r", "--name-only", REF], text=True).split())
out = []; gaps = cs.evaluation_gaps(11, tree, show, merged, anc, changed, out)
acc = sum(1 for c in json.loads(subprocess.check_output(["git","show",f"{REF}:{path}"],text=True))["cards"].values() if c.get("decision")=="accepted")
print(f"mode={mode} accepted={acc} gaps={len(gaps)} notmerged={sum('not merged' in g for g in gaps)} nobool={sum('no boolean gates_finders' in g for g in gaps)} parallel={len(out)}")
for g in gaps: print("GAP", g)
